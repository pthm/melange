{{- /* Syntax highlighting for ```fga blocks. Chroma has no OpenFGA lexer, so this tokenises each line with regexes into Chroma's own classes, which the theme's chroma CSS already colours in light and dark mode: k keywords (model, schema, type, relations, define, ...) nc type names after `type` nf relation names after `define` kt type restrictions inside [...] ow or, and, but not, from mf the schema version c1 # comments The wrapper mirrors Hextra's render-codeblock.html, so the copy button and `filename` attribute keep working. ponytail: regexes, not a parser; fine for the DSL's line-oriented syntax. */ -}} {{- $filename := .Attributes.filename | default "" -}} {{- $lines := slice -}} {{- range split (strings.TrimSuffix "\n" .Inner) "\n" -}} {{- $line := . | htmlEscape -}} {{- $comment := "" -}} {{- with findRESubmatch `^(.*?)((?:^|\s)#\s.*)$` $line 1 -}} {{- $line = index (index . 0) 1 -}} {{- $comment = index (index . 0) 2 -}} {{- end -}} {{- $line = replaceRE `^(\s*)type(\s+)([\w/.-]+)` `${1}type${2}${3}` $line -}} {{- $line = replaceRE `^(\s*)define(\s+)(\w+)` `${1}define${2}${3}` $line -}} {{- $line = replaceRE `^(\s*)schema(\s+)([\d.]+)` `${1}schema${2}${3}` $line -}} {{- $line = replaceRE `^(\s*)(model|relations|module|extend|condition)\b` `${1}${2}` $line -}} {{- $line = replaceRE `\[([^\]]*)\]` `[${1}]` $line -}} {{- $line = replaceRE `\b(but not|or|and|from|with)\b` `${1}` $line -}} {{- with $comment -}} {{- $line = printf `%s%s` $line . -}} {{- end -}} {{- $lines = $lines | append (printf `%s ` $line) -}} {{- end -}}
