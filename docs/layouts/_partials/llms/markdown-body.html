{{- /* A page's source as plain markdown for LLM agents: the .md twins and llms-full.txt. Shortcodes become their markdown equivalent (or are dropped when purely presentational), and relative links are made absolute, pointing at the target's .md twin when it has one, because a link like ../project-setup resolves differently from quick-start.md than from the HTML page it was written for. */ -}} {{- $page := . -}} {{- $c := .RawContent -}} {{- /* Tabs: label each tab's body. */ -}} {{- $c = replaceRE `\{\{< tab name="([^"]*)"[^>]*>\}\}` "**$1:**" $c -}} {{- /* Callouts: a labelled blockquote; the body continues it lazily. */ -}} {{- $c = replaceRE `\{\{< callout type="warning"[^>]*>\}\}` "> **Warning:**" $c -}} {{- $c = replaceRE `\{\{< callout type="(error|important)"[^>]*>\}\}` "> **Important:**" $c -}} {{- $c = replaceRE `\{\{< callout[^>]*>\}\}` "> **Note:**" $c -}} {{- /* Cards: a link list. */ -}} {{- $c = replaceRE `[ \t]*\{\{< card link="([^"]*)" title="([^"]*)" subtitle="([^"]*)"[^>]*>\}\}` "- [$2]($1): $3" $c -}} {{- $c = replaceRE `[ \t]*\{\{< card link="([^"]*)" title="([^"]*)"[^>]*>\}\}` "- [$2]($1)" $c -}} {{- /* Explain trees: the shortcode supplies the 
, so fence the body. */ -}}
{{- $c = replaceRE `\{\{< explaintree[^>]*>\}\}` "```text" $c -}}
{{- $c = replaceRE `\{\{< /explaintree >\}\}` "```" $c -}}

{{- /* Studio screenshots: the light capture. */ -}}
{{- $c = replaceRE `\{\{< studio-screenshot name="([^"]*)" alt="([^"]*)"[^>]*>\}\}` (printf "![$2](%s)" ("images/studio/$1-light.webp" | absURL)) $c -}}

{{- /* Anything left (tabs/cards wrappers, closing tags) is layout only. */ -}}
{{- $c = replaceRE `[ \t]*\{\{[<%][^}]*[>%]\}\}[ \t]*\n?` "" $c -}}

{{- /* Relative links. */ -}}
{{- range findRESubmatch `\]\(([^)\s]+)\)` $c -}}
  {{- $link := index . 1 -}}
  {{- if not (or (hasPrefix $link "http") (hasPrefix $link "mailto:") (hasPrefix $link "#")) -}}
    {{- $target := $link -}}
    {{- $anchor := "" -}}
    {{- with findRESubmatch `^([^#]*)(#.*)$` $link -}}
      {{- $target = index (index . 0) 1 -}}
      {{- $anchor = index (index . 0) 2 -}}
    {{- end -}}
    {{- if not (hasPrefix $target "/") -}}
      {{- $target = path.Join $page.RelPermalink $target -}}
    {{- end -}}
    {{- $url := $target | absURL -}}
    {{- if not (path.Ext $target) -}}
      {{- $url = printf "%s/" (strings.TrimSuffix "/" $target) | absURL -}}
    {{- end -}}
    {{- with site.GetPage $target -}}
      {{- with .OutputFormats.Get "markdown" -}}
        {{- $url = .Permalink -}}
      {{- end -}}
    {{- end -}}
    {{- $c = replace $c (printf "](%s)" $link) (printf "](%s%s)" $url $anchor) -}}
  {{- end -}}
{{- end -}}

{{- return $c -}}
